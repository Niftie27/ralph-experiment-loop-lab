// mock40.mjs: does m9-scan's requester/Budget count exactly the received body bytes of non-200 responses?
// Local node:http server on 127.0.0.1 only. Node builtins only. Run from the repo root.
import { createServer } from 'node:http';
import { resolve } from 'node:path';
import { pathToFileURL } from 'node:url';

const { requester, Budget, Abort } = await import(pathToFileURL(resolve('experiments/m9-gate/m9-scan.mjs')).href);

const hardStop = setTimeout(() => { process.stdout.write('timeout 55s\n'); process.exit(4); }, 55000);
hardStop.unref();

const KINDS = ['zip', 'sidecar'];
const PAIRS = [[340, 340], [349, 349], [379, 379], [349, 389], [379, 419]];
const MODES = ['chunked', 'length'];
const cases = [];
for (const kind of KINDS) for (const [a, b] of PAIRS) for (const mode of MODES) {
 const id = 'c' + String(cases.length + 1).padStart(2, '0');
 cases.push({ id, kind, a, b, mode, path: '/' + id + '/' + kind + '/' + a + '-' + b + '/' + mode });
}
const byPath = new Map(cases.map(c => [c.path, c]));
const hits = new Map();

// A body of exactly n bytes (ASCII), XML-like prefix, padded with 'x'.
function body(n) {
 const head = Buffer.from('<?xml version="1.0" encoding="UTF-8"?><Error>');
 const buf = Buffer.alloc(n, 0x78);
 head.copy(buf, 0, 0, Math.min(head.length, n));
 return buf;
}

const server = createServer((req, res) => {
 const c = byPath.get(req.url);
 const count = (hits.get(req.url) || 0) + 1;
 hits.set(req.url, count);
 const size = c ? (count === 1 ? c.a : c.b) : 0;
 const buf = body(size);
 const headers = { 'Content-Type': 'application/xml' };
 if (c && c.mode === 'length') headers['Content-Length'] = String(size);
 res.writeHead(404, headers);
 const half = Math.floor(size / 2);
 // Two writes; the short delay keeps them as separate writes on the wire.
 res.write(buf.subarray(0, half));
 setTimeout(() => res.end(buf.subarray(half)), 5);
});

await new Promise((ok, fail) => { server.once('error', fail); server.listen(0, '127.0.0.1', ok); });
const port = server.address().port;

const lines = [];
let allMatch = true;
for (const c of cases) {
 const budget = new Budget();
 const get = requester({ pause: 0, retry: 0 }, budget);
 const url = 'http://127.0.0.1:' + port + c.path;
 let result;
 try {
  const r = await get(c.kind, url);
  result = 'status:' + r.status + (r.body === null ? '' : ' body:' + r.body.length);
 } catch (e) {
  if (e instanceof Abort) result = 'abort:' + e.reason;
  else result = 'error:' + (e && e.message);
 }
 const expected = c.a + c.b;
 const match = budget.total === expected;
 if (!match) allMatch = false;
 lines.push([c.kind, c.a, c.b, c.mode, hits.get(c.path) || 0, budget.total, expected, match, result].join('|'));
}
for (const l of lines) process.stdout.write(l + '\n');
process.stdout.write('all match ' + allMatch + '\n');

server.closeAllConnections();
await new Promise(ok => server.close(ok));
clearTimeout(hardStop);
